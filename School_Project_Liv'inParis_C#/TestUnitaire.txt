
using Microsoft.VisualStudio.TestTools.UnitTesting;

[TestClass]
public class MetroFormTests
{
    [TestMethod]
    public void Constructor_ShouldInitializeCorrectly()
    {
        // Arrange
        var parcoursPCM = new List<int> { 1, 2, 3, 4 };

        // Act
        var metroForm = new MetroForm(parcoursPCM);

        // Assert
        Assert.IsNotNull(metroForm);  // Vérifie si l'objet MetroForm est correctement créé
        Assert.AreEqual(4, metroForm.parcoursPCM.Count);  // Vérifie la taille du parcoursPCM
        Assert.AreEqual(6, metroForm.pcmSegments.Count);  // Vérifie que 6 segments ont été ajoutés (2 sens pour chaque paire)
    }
[TestMethod]
public void InitializeDrawingResources_ShouldInitializePens()
{
    // Arrange
    var parcoursPCM = new List<int> { 1, 2, 3, 4 };
    var metroForm = new MetroForm(parcoursPCM);

    // Act
    metroForm.InitializeDrawingResources();

    // Assert
    Assert.IsTrue(metroForm.linePens.ContainsKey("DEFAULT"));  // Vérifie que le pen "DEFAULT" a été ajouté
    Assert.AreEqual(2.0f, metroForm.linePens["DEFAULT"].Width); // Vérifie que la largeur du pen est correcte
}

[TestMethod]
public void LoadStations_ShouldLoadStationsCorrectly()
{
    // Arrange
    var parcoursPCM = new List<int> { 1, 2, 3, 4 };
    var metroForm = new MetroForm(parcoursPCM);
    string testFilePath = "testNoeuds.csv"; // Fichier CSV simulé

    // Mocking File.ReadLines to simulate file reading
    var fakeFileContent = new List<string>
    {
        "ID;Line;Name;Lon;Lat",
        "1;Line1;Station1;2.35;48.85",
        "2;Line1;Station2;2.36;48.86"
    };

    File.WriteAllLines(testFilePath, fakeFileContent);

    // Act
    metroForm.LoadStations(testFilePath, Encoding.UTF8);

    // Assert
    Assert.IsTrue(metroForm.stations.ContainsKey(1));  // Vérifie si la station avec ID 1 existe
    Assert.AreEqual("Station1", metroForm.stations[1].Name);  // Vérifie le nom de la station
}
}
    [TestClass]
    public class GrapheTests
    {
        private const string TestFilePath = "testfile.csv";
        private const string InvalidFilePath = "invalidfile.csv";

        private static readonly string[] TestFileContent =
        {
            "StationID;StationName;PreviousStation;NextStation;Time;Correspondances",
            "0;StationA;;1;5;",
            "1;StationB;0;2;5;3 10",
            "2;StationC;1;;5;",
            "3;StationD;;;0;1 10",
            "4;IsolatedStation;;;0;"
        };

	[TestInitialize]
        public void Setup() => File.WriteAllLines(TestFilePath, TestFileContent);


        [TestCleanup]
        public void Cleanup()
        {
            if (File.Exists(TestFilePath))
                File.Delete(TestFilePath);
        }

        [TestMethod]
        public void Initialisation_Graphe_AvecFichierValide_DoitCreerMatriceEtDictionnaire()
        {
            // Act
            var graphe = new Graphe(TestFilePath);

            // Assert
            Assert.AreEqual(333, graphe.GRAPHE.GetLength(0));
            Assert.AreEqual(333, graphe.GRAPHE.GetLength(1));
            Assert.AreEqual("StationA", GetKeyFromValue(graphe.DicoIDNOM, 0));
            Assert.AreEqual(5, graphe.GRAPHE[0, 1]);
            Assert.AreEqual(10, graphe.GRAPHE[1, 3]);
        }

        [TestMethod]
        [ExpectedException(typeof(FileNotFoundException))]
        public void Initialisation_Graphe_AvecFichierInvalide_DoitLeverException()
        {
            // Act
            new Graphe(InvalidFilePath);
        }

        [TestMethod]
        public void CalculerPoidsChemin_CheminValide_DoitRetournerPoidsCorrect()
        {
            // Arrange
            var graphe = new Graphe(TestFilePath);
            var chemin = new List<int> { 0, 1, 2 };

            // Act
            var poids = graphe.CalculerPoidsChemin(chemin);

            // Assert
            Assert.AreEqual(10, poids);
        }

        [TestMethod]
        public void TrouverCheminDjikstra_CheminSimple_DoitRetournerCheminOptimal()
        {
            // Arrange
            var graphe = new Graphe(TestFilePath);

            // Act
            var chemin = graphe.TrouverCheminDjikstra(0, 3);

            // Assert
            CollectionAssert.AreEqual(new List<int> { 0, 1, 3 }, chemin);
            Assert.AreEqual(15, graphe.CalculerPoidsChemin(chemin));
        }

        [TestMethod]
        [ExpectedException(typeof(Exception))]
        public void TrouverCheminDjikstra_NoeudIsole_DoitLeverException()
        {
            // Arrange
            var graphe = new Graphe(TestFilePath);

            // Act
            graphe.TrouverCheminDjikstra(0, 4);
        }

        [TestMethod]
        public void TrouverCheminBellmanFord_CheminAvecCorrespondance_DoitRetournerChemin()
        {
            // Arrange
            var graphe = new Graphe(TestFilePath);

            // Act
            var chemin = graphe.TrouverCheminBellmanFord(3, 2);

            // Assert
            CollectionAssert.AreEqual(new List<int> { 3, 1, 2 }, chemin);
            Assert.AreEqual(15, graphe.CalculerPoidsChemin(chemin));
        }

        [TestMethod]
        public void TrouverCheminFloydWarshall_CheminInverse_DoitRetournerChemin()
        {
            // Arrange
            var graphe = new Graphe(TestFilePath);

            // Act
            var chemin = graphe.TrouverCheminFloydWarshall(2, 0);

            // Assert
            CollectionAssert.AreEqual(new List<int> { 2, 1, 0 }, chemin);
            Assert.AreEqual(10, graphe.CalculerPoidsChemin(chemin));
        }

        // Méthode utilitaire pour rechercher dans le dictionnaire
        private string GetKeyFromValue(Dictionary<string, int> dict, int value)
        {
            foreach (var pair in dict)
                if (pair.Value == value)
                    return pair.Key;
            return null;
        }
    }
